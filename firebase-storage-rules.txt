// Firebase Storage Rules for HappyDog2
// Firebase Console → Storage → Rules 탭에서 다음 규칙을 복사해서 붙여넣으세요

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // posts 폴더: 공개 읽기, 로그인한 사용자만 자신의 폴더에 쓰기
    match /posts/{userId}/{allPaths=**} {
      allow read: if true;  // 누구나 읽기 가능 (공개 이미지)
      allow write: if request.auth != null && request.auth.uid == userId;
    }
    
    // cartoon_source 폴더: 공개 읽기, 로그인한 사용자만 자신의 폴더에 쓰기
    match /cartoon_source/{userId}/{allPaths=**} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == userId;
    }
    
    // cartoon_result 폴더: 공개 읽기, 로그인한 사용자만 자신의 폴더에 쓰기
    match /cartoon_result/{userId}/{allPaths=**} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == userId;
    }
    
    // profile_images 폴더: 공개 읽기, 로그인한 사용자만 자신의 폴더에 쓰기
    match /profile_images/{userId}/{allPaths=**} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == userId;
    }
    
    // 기타 모든 파일: 로그인 필요
    match /{allPaths=**} {
      allow read, write: if request.auth != null;
    }
  }
}
